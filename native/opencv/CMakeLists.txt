cmake_minimum_required(VERSION 3.22)
project(UniversalStudioVision LANGUAGES CXX)
find_package(OpenCV 5.0.0 EXACT CONFIG REQUIRED COMPONENTS core geometry imgproc)
add_library(ust_cv SHARED adapter.cpp)
target_compile_features(ust_cv PRIVATE cxx_std_17)
target_include_directories(ust_cv PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ust_cv PRIVATE ${OpenCV_LIBS})
set_target_properties(ust_cv PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(ANDROID)
  target_link_options(ust_cv PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(NOT MSVC)
  target_link_options(ust_cv PRIVATE "-Wl,--gc-sections" "-Wl,--exclude-libs,ALL")
endif()
